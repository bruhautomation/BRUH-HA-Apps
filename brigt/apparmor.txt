#include <tunables/global>

# AppArmor profile for the BRigt add-on.
#
# The threat this profile is for
# ------------------------------
# BRigt runs with `host_network: true` (LIFX discovery is a UDP broadcast,
# and cue latency is the product), which puts the container on the LAN
# itself — so the network-facing denials here are worth more than they
# would be on a bridged add-on. The profile does not try to constrain what
# BRigt does inside its container (it reads music, writes /data, sends UDP
# to bulbs, calls the Supervisor API); it denies the host-escape
# primitives: mounting, kernel modules, raw sockets, kernel tunables, and
# the Docker socket.

profile brigt flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # ------------------------------------------------------------------
  # Host-escape primitives — denied outright
  # ------------------------------------------------------------------

  # Kernel tunables. Nothing in this add-on tunes the host kernel.
  deny /proc/sys/** wklx,
  deny /sys/kernel/** wklx,
  deny @{PROC}/sysrq-trigger rwklx,

  # Kernel module loading and raw device access.
  deny /sys/module/** wklx,
  deny /dev/mem rwklx,
  deny /dev/kmem rwklx,
  deny /dev/port rwklx,
  deny /dev/kmsg rwklx,

  # Firmware and boot.
  deny /sys/firmware/** rwklx,
  deny /boot/** rwklx,

  # The Docker socket. BRigt talks to the Supervisor over its HTTP API with
  # SUPERVISOR_TOKEN; a writable docker.sock is root on the host by another
  # name.
  deny /var/run/docker.sock rwklx,
  deny /run/docker.sock rwklx,

  # ------------------------------------------------------------------
  # Capabilities
  # ------------------------------------------------------------------
  # run.sh creates the `brigt` user (UID 1000) and drops to it with
  # su-exec, so setuid/setgid and chown are genuinely needed. Everything
  # dangerous is absent by omission and then denied explicitly so a future
  # edit has to be deliberate. LIFX control is ordinary UDP datagrams —
  # net_raw stays denied.
  capability chown,
  capability dac_override,
  capability fowner,
  capability fsetid,
  capability kill,
  capability setgid,
  capability setuid,
  capability setpcap,
  capability audit_write,

  deny capability sys_admin,
  deny capability sys_module,
  deny capability sys_rawio,
  deny capability sys_ptrace,
  deny capability sys_boot,
  deny capability net_admin,
  deny capability net_raw,
  deny capability mac_admin,
  deny capability mac_override,
  deny capability dac_read_search,

  # No mounting, ever.
  deny mount,
  deny umount,
  deny pivot_root,

  # ------------------------------------------------------------------
  # What the add-on legitimately does
  # ------------------------------------------------------------------

  # Filesystem: /config (integration deploy + IPC), /media (the music),
  # /data, and the container's own tree.
  file,

  # Signals: run.sh's cleanup trap tears down the bridge and the panel on
  # SIGTERM; the panel signals its own workers.
  signal (send,receive) set=(kill,term,int,hup,cont,usr1,usr2,winch),

  ptrace (trace,read) peer=brigt,

  # Networking: the panel (8095), UDP 56700 to the bulbs (broadcast and
  # unicast), the Supervisor API, and outbound HTTPS for lyrics lookups.
  # `network netlink raw` is interface enumeration, not `network raw` —
  # raw sockets stay denied via the net_raw capability above.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network unix stream,
  network unix dgram,
  network netlink raw,

  # Anything the add-on execs runs under this same profile rather than
  # escaping to the unconfined host profile.
  /** ix,
}
